using TenantDesk.Application.Interfaces;
using TenantDesk.Application.Interfaces.Services;
using TenantDesk.Domain.Entities;

namespace TenantDesk.Application.Services;

public class SlaCalculatorService(ITicketRepository _ticketRepository, ISlaRepository _slaRepository, IBusinessHourRepository _businessHourRepository, IDateTimeProvider _dateTimeProvider) : ISlaCalculatorService
{
    

    private const int MaxDaySearchWindow = 365;

    public async Task<DateTime> CalculateResponseDueAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        var sla = await GetSlaOrThrowAsync(ticket);

        return await CalculateDueDateAsync(ticket, sla.ResponseTime);
    }

    public async Task<DateTime> CalculateResolutionDueAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        return await CalculateResolutionDueInternalAsync(ticket);
    }

    public async Task<bool> IsBreachedAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        var resolutionDue = await CalculateResolutionDueInternalAsync(ticket);

        var comparisonTime = ticket.Status == TicketStatus.Closed
            ? ticket.UpdatedAt
            : _dateTimeProvider.UtcNow;

        return comparisonTime > resolutionDue;
    }

    private async Task<DateTime> CalculateResolutionDueInternalAsync(Ticket ticket)
    {
        var sla = await GetSlaOrThrowAsync(ticket);
        return await CalculateDueDateAsync(ticket, sla.ResolutionTime);
    }

    private async Task<DateTime> CalculateDueDateAsync(Ticket ticket, int minutesToAllocate)
    {
        var businessHours = await _businessHourRepository.GetByTenantAsync(ticket.TenantId);

        var workingDaysMap = businessHours
            .Where(b => b.IsWorkingDay)
            .ToDictionary(b => b.DayOfWeek);

        if (workingDaysMap.Count == 0)
            return ticket.CreatedAt.AddMinutes(minutesToAllocate);

        var currentPointer = ticket.CreatedAt;
        var remainingTime = TimeSpan.FromMinutes(minutesToAllocate);

        for (var i = 0; i < MaxDaySearchWindow; i++)
        {
            if (!workingDaysMap.TryGetValue(currentPointer.DayOfWeek, out var todayHours))
            {
                currentPointer = currentPointer.Date.AddDays(1);
                continue;
            }

            var openTime = todayHours.OpenTime!.Value.ToTimeSpan();
            var closeTime = todayHours.CloseTime!.Value.ToTimeSpan();
            var currentTimeOfDay = currentPointer.TimeOfDay;

            if (currentTimeOfDay < openTime)
            {
                currentPointer = currentPointer.Date.Add(openTime);
                currentTimeOfDay = openTime;
            }

            if (currentTimeOfDay >= closeTime)
            {
                currentPointer = currentPointer.Date.AddDays(1);
                continue;
            }

            var availableTimeToday = closeTime - currentTimeOfDay;

            if (remainingTime <= availableTimeToday)
                return currentPointer.Add(remainingTime);

            remainingTime -= availableTimeToday;
            currentPointer = currentPointer.Date.AddDays(1);
        }

        throw new InvalidOperationException("Could not calculate SLA due date within the maximum search window (365 days). Check the tenant's business hour configuration.");
    }

    private async Task<Ticket> GetTicketOrThrowAsync(Guid ticketId)
    {
        var ticket = await _ticketRepository.GetByIdAsync(ticketId);
        if (ticket is null)
            throw new KeyNotFoundException($"Ticket '{ticketId}' not found.");

        return ticket;
    }

    private async Task<Sla> GetSlaOrThrowAsync(Ticket ticket)
    {
        var sla = await _slaRepository.GetByIdAsync(ticket.SlaId);
        if (sla is null)
            throw new KeyNotFoundException($"SLA '{ticket.SlaId}' referenced by ticket '{ticket.TicketId}' was not found.");

        return sla;
    }

using AutoMapper;
using TenantDesk.Application.Dto.Sla;
using TenantDesk.Application.Interfaces;
using TenantDesk.Application.Interfaces.Identity;
using TenantDesk.Domain.Entities;

namespace TenantDesk.Application.Services;

public class SlaService
{
    private readonly ISlaRepository _slaRepository;
    private readonly ICurrentTenantService _currentTenantService;
    private readonly IMapper _mapper;

    public SlaService(ISlaRepository slaRepository,ICurrentTenantService currentTenantService,IMapper mapper)
    {
        _slaRepository = slaRepository;
        _currentTenantService = currentTenantService;
        _mapper = mapper;
    }

    public async Task<SlaResponseDto> CreateAsync(CreateSlaDto dto)
    {
        if (dto.ResponseTime > dto.ResolutionTime)
            throw new ArgumentException("Response time cannot be greater than resolution time.");

        var tenantId = _currentTenantService.GetTenantId();

        var existing = await _slaRepository.GetByTenantAndPriorityAsync(tenantId, dto.Priority);
        if (existing is not null)
            throw new InvalidOperationException($"An SLA for priority '{dto.Priority}' already exists for this tenant.");

        var sla = new Sla(tenantId, dto.Priority, dto.ResponseTime, dto.ResolutionTime);
        await _slaRepository.AddAsync(sla);

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task<SlaResponseDto> GetByIdAsync(Guid id)
    {
        var sla = await _slaRepository.GetByIdAsync(id);
        EnsureExistsAndOwned(sla);

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task<SlaResponseDto> GetByPriorityAsync(PriorityLevel priority)
    {
        var tenantId = _currentTenantService.GetTenantId();
       
        var sla = await _slaRepository.GetByTenantAndPriorityAsync(tenantId, priority);

        if (sla == null){
            throw new KeyNotFoundException($"No SLA configured for priority '{priority}'.");
        }

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task<IEnumerable<SlaResponseDto>> GetAllAsync()
    {
        var tenantId = _currentTenantService.GetTenantId();
        var slas = await _slaRepository.GetAllByTenantAsync(tenantId);

        return _mapper.Map<IEnumerable<SlaResponseDto>>(slas);
    }

    public async Task<SlaResponseDto> UpdateAsync(Guid id, UpdateSlaTimesDto dto)
    {
        if (dto.ResponseTime > dto.ResolutionTime)
            throw new ArgumentException("Response time cannot be greater than resolution time.");

        var sla = await _slaRepository.GetByIdAsync(id);
        EnsureExistsAndOwned(sla);

        sla.UpdateTimes(dto.ResponseTime, dto.ResolutionTime);
        await _slaRepository.UpdateAsync(sla);

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task DeleteAsync(Guid id)
    {
        var sla = await _slaRepository.GetByIdAsync(id);
        EnsureExistsAndOwned(sla);

        await _slaRepository.DeleteAsync(id);
    }

    private void EnsureExistsAndOwned([NotNull] Sla? sla)
    {
        if (sla is null)
            throw new KeyNotFoundException("SLA not found.");

        if (sla.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot access resources belonging to another tenant.");
    }

    public async Task<UserResponseDto> GetByIdAsync(Guid id)

        // tenant scoping: ما بنسمح تشوف مستخدم tenant تاني
        if (user.TenantId != _currentTenantService.GetTenantId())

    public async Task<PagedResult<UserResponseDto>> GetFilteredAsync(UserFilter filter)

        filter.TenantId = _currentTenantService.GetTenantId();

        return new PagedResult<UserResponseDto>
        {
            Items = result.Items.Select(MapToResponseDto).ToList(),
            TotalCount = result.TotalCount,
            Page = result.Page,
            PageSize = result.PageSize
        };
    }

    public async Task<UserResponseDto> UpdateAsync(Guid id, UpdateUserDto dto)
    {
        var user = await _userRepository.GetByIdAsync(id)
            ?? throw new KeyNotFoundException($"User with ID '{id}' not found.");

        if (user.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot modify users from another tenant.");

        var normalizedEmail = dto.Email.Trim().ToLower();

        if (!string.Equals(user.Email, normalizedEmail, StringComparison.OrdinalIgnoreCase))
        {
            var emailTaken = await _userRepository.ExistsByEmailAsync(user.TenantId, normalizedEmail);
            if (emailTaken)
                throw new InvalidOperationException("A user with this email already exists in this tenant.");
        }

        user.ChangeEmail(normalizedEmail);
        user.ChangeRole(dto.Role);

        await _userRepository.UpdateAsync(user);

        return _mapper.Map<UserResponseDto>(user);
    }

    public async Task ChangeStatusAsync(Guid id, ChangeUserStatusDto dto)
    {
        var user = await _userRepository.GetByIdAsync(id)

        if (dto.IsActive)

    }

            ?? throw new KeyNotFoundException($"User with ID '{id}' not found.");

        if (user.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot delete users from another tenant.");

        await _userRepository.DeleteAsync(id);
    }

        var ticket = new Ticket(tenantId, dto.CategoryId, dto.SlaId, userId, dto.Title, dto.Description);

        await _ticketRepository.AddAsync(ticket);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    public async Task<TicketResponseDto> GetByIdAsync(Guid id)
    {
        var ticket = await GetOwnedTicketAsync(id);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    public async Task<PagedResult<TicketResponseDto>> GetFilteredAsync(TicketFilter filter)
    {
        filter.TenantId = _currentTenantService.GetTenantId();

        // Customers only ever see their own tickets, regardless of what they pass in the query.
        if (_currentUserService.GetUserRole() == UserRole.Customer)
            filter.CreatedBy = _currentUserService.GetUserId();

        var result = await _ticketRepository.GetFilteredAsync(filter);

        return new PagedResult<TicketResponseDto>
        {
            Items = result.Items.Select(t => _mapper.Map<TicketResponseDto>(t)).ToList(),
            TotalCount = result.TotalCount,
            Page = result.Page,
            PageSize = result.PageSize
        };
    }

    public async Task<TicketResponseDto> UpdateDetailsAsync(Guid id, UpdateTicketDto dto)
    {
        var ticket = await GetOwnedTicketAsync(id, agentOrAdminOnly: true);

        ticket.UpdateDetails(dto.Title, dto.Description);

        await _ticketRepository.UpdateAsync(ticket);

        await _notificationDispatcher.DispatchAsync(
            ticket.CreatedBy, NotificationType.TicketUpdated, $"Your ticket '{ticket.Title}' was updated.", ticket.TicketId);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    public async Task<TicketResponseDto> ChangeStatusAsync(Guid id, TicketStatus newStatus)
    {
        var ticket = await GetOwnedTicketAsync(id, agentOrAdminOnly: true);

        ticket.ChangeStatus(newStatus);

        await _ticketRepository.UpdateAsync(ticket);

        await _notificationDispatcher.DispatchAsync(
            ticket.CreatedBy, NotificationType.TicketUpdated, $"Your ticket '{ticket.Title}' status changed to {newStatus}.", ticket.TicketId);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    public async Task<TicketResponseDto> AssignAsync(Guid id, Guid assigneeUserId)
    {
        var ticket = await GetOwnedTicketAsync(id, agentOrAdminOnly: true);

        ticket.AssignTo(assigneeUserId);

        await _ticketRepository.UpdateAsync(ticket);

        await _notificationDispatcher.DispatchAsync(
            assigneeUserId, NotificationType.TicketAssigned, $"You have been assigned ticket '{ticket.Title}'.", ticket.TicketId);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    public async Task<TicketResponseDto> CloseAsync(Guid id)
    {
        // Agent/Admin can close any ticket assigned to them; a Customer can only close their own.
        var ticket = await GetOwnedTicketAsync(id);

        ticket.Close();

        await _ticketRepository.UpdateAsync(ticket);

        await _notificationDispatcher.DispatchAsync(
            ticket.CreatedBy, NotificationType.TicketUpdated, $"Your ticket '{ticket.Title}' was closed.", ticket.TicketId);

        return _mapper.Map<TicketResponseDto>(ticket);
    }

    private async Task<Ticket> GetOwnedTicketAsync(Guid id, bool agentOrAdminOnly = false)
    {
        var ticket = await _ticketRepository.GetByIdAsync(id)
            ?? throw new KeyNotFoundException($"Ticket with ID '{id}' not found.");

        if (ticket.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot access tickets from another tenant.");

        var role = _currentUserService.GetUserRole();
        var userId = _currentUserService.GetUserId();

        if (role == UserRole.Customer)
        {
            if (ticket.CreatedBy != userId)
                throw new UnauthorizedAccessException("You cannot access a ticket you did not create.");

            if (agentOrAdminOnly)
                throw new UnauthorizedAccessException("Only an agent or admin can perform this action.");
        }
        else if (role == UserRole.Agent && agentOrAdminOnly)
        {
            // Agent can only modify tickets assigned to them; Admin is exempt from this check entirely.
            if (ticket.AssignedTo != userId)
                throw new UnauthorizedAccessException("You can only modify tickets assigned to you.");
        }

        return ticket;
    }

    public Guid TenantId { get; private set; }

    public string Subdomain { get; private set; } = null!;
    public bool IsActive { get; private set; }
    public DateTime CreatedAt { get; private set; }
    public DateTime? UpdatedAt { get; private set; }

    public Plan Plan { get; private set; } = null!;

    private Tenant() { }

    public Tenant(Guid planId, string tenantName, string subdomain)
    {
        
        if (planId == Guid.Empty)
            throw new ArgumentException("Plan ID cannot be empty.", nameof(planId));

        if (string.IsNullOrWhiteSpace(tenantName))
            throw new ArgumentException("Tenant name cannot be null or empty.", nameof(tenantName));

        if (string.IsNullOrWhiteSpace(subdomain))
            throw new ArgumentException("Subdomain cannot be null or empty.", nameof(subdomain));

        PlanId = planId;
        TenantName = tenantName.Trim();
        Subdomain = subdomain.Trim().ToLower();
        IsActive = true;
        CreatedAt = DateTime.UtcNow;
    }

    public void Rename(string name)
    {
        if (string.IsNullOrWhiteSpace(name))
            throw new ArgumentException("Tenant name cannot be null or empty.", nameof(name));

        TenantName = name.Trim();
        UpdateTimestamp();
    }

    public void ChangePlan(Guid planId)
    {
        if (planId == Guid.Empty)
            throw new ArgumentException("Plan ID cannot be empty.", nameof(planId));

        PlanId = planId;
        UpdateTimestamp();
    }

    public void ChangeSubdomain(string subdomain)
    {
        if (string.IsNullOrWhiteSpace(subdomain))
            throw new ArgumentException("Subdomain cannot be null or empty.", nameof(subdomain));

        Subdomain = subdomain.ToLowerInvariant().Trim();
        UpdateTimestamp();
    }

    public void Activate()
    {
        if (!IsActive)
        {
            IsActive = true;
            UpdateTimestamp();
        }
    }

    public void Deactivate()
    {
        if (IsActive)
        {
            IsActive = false;
            UpdateTimestamp();
        }
    }

    }

namespace TenantDesk.Api.Middleware;

public class ExceptionMiddleware(
    RequestDelegate next,
    ILogger<ExceptionMiddleware> logger)
{
    private readonly RequestDelegate _next = next;
    private readonly ILogger<ExceptionMiddleware> _logger = logger;

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unhandled exception");

            context.Response.ContentType = "application/json";

            context.Response.StatusCode = ex switch
            {
                KeyNotFoundException => StatusCodes.Status404NotFound,

                UnauthorizedAccessException => StatusCodes.Status403Forbidden,

                InvalidOperationException => StatusCodes.Status409Conflict,

                _ => StatusCodes.Status400BadRequest
            };

            await context.Response.WriteAsJsonAsync(new
            {
                error = ex.Message
            });
        }
    }

using TenantDesk.Application.Common.Interfaces;
using Microsoft.EntityFrameworkCore.Storage;

namespace TenantDesk.Infrastructure.Data;

public class UnitOfWork(AppDbContext context) : IUnitOfWork
{
    private readonly AppDbContext _context = context;
    private IDbContextTransaction? _transaction;

    public Task<int> SaveChangesAsync() => _context.SaveChangesAsync();

    public async Task BeginTransactionAsync() =>
        _transaction = await _context.Database.BeginTransactionAsync();

    public async Task CommitAsync()
    {
        if (_transaction is null) return;

        await _context.SaveChangesAsync();
        await _transaction.CommitAsync();
        await _transaction.DisposeAsync();

        _transaction = null;
    }

    public async Task RollbackAsync()
    {
        if (_transaction is null) return;

        await _transaction.RollbackAsync();
        await _transaction.DisposeAsync();

        _transaction = null;
    }

using Microsoft.AspNetCore.Identity;
using TenantDesk.Application.Interfaces;
using TenantDesk.Application.Interfaces.Identity;
using TenantDesk.Application.Common.Interfaces;
using TenantDesk.Application.Dto.Auth;

using TenantDesk.Domain.Enums;

namespace TenantDesk.Infrastructure.Identity;

public class AuthService(
    IUserRepository userRepository,
    ITenantRepository _tenantRepository,
    ITokenService _tokenService,
    IPasswordHasher<User> _passwordHasher,
    IUnitOfWork _unitOfWork) : IAuthService
{
    

    public async Task<AuthResponseDto> RegisterAsync(RegisterDto dto)
    {
        // 1. Resolve the tenant from the subdomain the frontend sent.
        //    If it doesn't exist, we can't know which tenant to attach
        //    this user to, so we stop here.
        var tenant = await _tenantRepository.GetBySubdomainAsync(dto.TenantSubdomain)
            ?? throw new KeyNotFoundException("Tenant not found.");

        // 2. Prevent duplicate accounts WITHIN this tenant. Per-tenant
        //    uniqueness means the same email CAN exist in different tenants,
        //    but not twice in the same one — that's exactly what
        //    ExistsByEmailAsync(tenantId, email) checks.
        var emailTaken = await userRepository.ExistsByEmailAsync(tenant.TenantId, dto.Email.Trim().ToLower());
        if (emailTaken)
            throw new InvalidOperationException("A user with this email already exists in this tenant.");

        // 3. Create the user WITHOUT a password (see Step 2 change to User.cs),
        //    then hash the real password and attach it explicitly.
        //    Default role here is Customer — change this if your business
        //    rule is different (e.g. invite-only admins vs self-signup customers).
        var user = new User(tenant.TenantId, dto.Email, UserRole.Customer);
        var hash = _passwordHasher.HashPassword(user, dto.Password);
        user.SetPassword(hash);

        // 4. Persist to the database. AddAsync stages the change in the
        //    DbContext; the actual SQL INSERT + commit happens in
        //    _unitOfWork.SaveChangesAsync() — this is the pattern your
        //    UnitOfWork class already uses everywhere else in the project.
        await userRepository.AddAsync(user);
        await _unitOfWork.SaveChangesAsync();

        // 5. Issue a token immediately so the user is logged in right after
        //    registering (no separate login step required).
        var token = _tokenService.TokenGenerateToken(user);
        var expiresAt = GetExpiryFromToken(token);
        return new AuthResponseDto(token, user.UserId, user.TenantId, user.Email, user.Role, expiresAt);
    }

    public async Task<AuthResponseDto> LoginAsync(LoginDto dto)
    {
        // Resolve tenant the same way as registration.
        var tenant = await _tenantRepository.GetBySubdomainAsync(dto.TenantSubdomain)
            ?? throw new UnauthorizedAccessException("Invalid credentials.");

        // Tenant-scoped lookup: finds the user only if they belong to THIS
        // tenant, so there's no ambiguity if the same email exists under a
        // different tenant.
        var user = await userRepository.GetByEmailAsync(dto.Email.Trim().ToLower(), tenant.TenantId)
            ?? throw new UnauthorizedAccessException("Invalid credentials.");

        // Note: we deliberately throw the SAME generic "Invalid credentials"
        // message whether the tenant, email, or password is wrong. This
        // stops attackers from figuring out (e.g.) "this email exists but
        // the password is wrong" — a common security leak in login endpoints.

        if (!user.IsActive)
            throw new UnauthorizedAccessException("This account has been deactivated.");

        var result = _passwordHasher.VerifyHashedPassword(user, user.PasswordHash, dto.Password);
        if (result == PasswordVerificationResult.Failed)
            throw new UnauthorizedAccessException("Invalid credentials.");

        var token = _tokenService.TokenGenerateToken(user);
        var expiresAt = GetExpiryFromToken(token);
        return new AuthResponseDto(token, user.UserId, user.TenantId, user.Email, user.Role, expiresAt);
    }

    // ITokenService only returns the raw JWT string, not an expiry value
    // separately. Rather than recompute/duplicate the expiry duration here
    // (which could drift out of sync with whatever TokenService actually
    // set inside the token), we read it straight out of the token's own
    // "exp" claim. The token itself stays the single source of truth.
    private static DateTime GetExpiryFromToken(string token)
    {
        var handler = new JwtSecurityTokenHandler();
        var jwt = handler.ReadJwtToken(token);
        return jwt.ValidTo; 
    }

public class TokenService(IConfiguration config) : ITokenService
{
    private readonly IConfiguration _config = config;

    public string TokenGenerateToken(User user)    {
        var jwtSection = _config.GetSection("Jwt");

        
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtSection["Key"]!));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        
        var expiresAt = DateTime.UtcNow.AddHours(8);

       
        var claims = new List<Claim>
        {
            new Claim("local_user_id", user.UserId.ToString()),
            new Claim("tenant_id", user.TenantId.ToString()),
            new Claim(ClaimTypes.Email, user.Email),
            new Claim(ClaimTypes.Role, user.Role.ToString())
        };
      

        var token = new JwtSecurityToken(
            issuer: jwtSection["Issuer"],
            audience: jwtSection["Audience"],
            claims: claims,
            expires: expiresAt,
            signingCredentials: creds
        );

       
        var tokenString = new JwtSecurityTokenHandler().WriteToken(token);

        return  tokenString;;
    }
