using TenantDesk.Application.Interfaces;
using TenantDesk.Application.Interfaces.Services;
using TenantDesk.Domain.Entities;

namespace TenantDesk.Application.Services;

public class SlaCalculatorService(ITicketRepository _ticketRepository, ISlaRepository _slaRepository, IBusinessHourRepository _businessHourRepository, IDateTimeProvider _dateTimeProvider) : ISlaCalculatorService
{
    

    private const int MaxDaySearchWindow = 365;

    public async Task<DateTime> CalculateResponseDueAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        var sla = await GetSlaOrThrowAsync(ticket);

        return await CalculateDueDateAsync(ticket, sla.ResponseTime);
    }

    public async Task<DateTime> CalculateResolutionDueAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        return await CalculateResolutionDueInternalAsync(ticket);
    }

    public async Task<bool> IsBreachedAsync(Guid ticketId)
    {
        var ticket = await GetTicketOrThrowAsync(ticketId);
        var resolutionDue = await CalculateResolutionDueInternalAsync(ticket);

        var comparisonTime = ticket.Status == TicketStatus.Closed
            ? ticket.UpdatedAt
            : _dateTimeProvider.UtcNow;

        return comparisonTime > resolutionDue;
    }

    private async Task<DateTime> CalculateResolutionDueInternalAsync(Ticket ticket)
    {
        var sla = await GetSlaOrThrowAsync(ticket);
        return await CalculateDueDateAsync(ticket, sla.ResolutionTime);
    }

    private async Task<DateTime> CalculateDueDateAsync(Ticket ticket, int minutesToAllocate)
    {
        var businessHours = await _businessHourRepository.GetByTenantAsync(ticket.TenantId);

        var workingDaysMap = businessHours
            .Where(b => b.IsWorkingDay)
            .ToDictionary(b => b.DayOfWeek);

        if (workingDaysMap.Count == 0)
            return ticket.CreatedAt.AddMinutes(minutesToAllocate);

        var currentPointer = ticket.CreatedAt;
        var remainingTime = TimeSpan.FromMinutes(minutesToAllocate);

        for (var i = 0; i < MaxDaySearchWindow; i++)
        {
            if (!workingDaysMap.TryGetValue(currentPointer.DayOfWeek, out var todayHours))
            {
                currentPointer = currentPointer.Date.AddDays(1);
                continue;
            }

            var openTime = todayHours.OpenTime!.Value.ToTimeSpan();
            var closeTime = todayHours.CloseTime!.Value.ToTimeSpan();
            var currentTimeOfDay = currentPointer.TimeOfDay;

            if (currentTimeOfDay < openTime)
            {
                currentPointer = currentPointer.Date.Add(openTime);
                currentTimeOfDay = openTime;
            }

            if (currentTimeOfDay >= closeTime)
            {
                currentPointer = currentPointer.Date.AddDays(1);
                continue;
            }

            var availableTimeToday = closeTime - currentTimeOfDay;

            if (remainingTime <= availableTimeToday)
                return currentPointer.Add(remainingTime);

            remainingTime -= availableTimeToday;
            currentPointer = currentPointer.Date.AddDays(1);
        }

        throw new InvalidOperationException("Could not calculate SLA due date within the maximum search window (365 days). Check the tenant's business hour configuration.");
    }

    private async Task<Ticket> GetTicketOrThrowAsync(Guid ticketId)
    {
        var ticket = await _ticketRepository.GetByIdAsync(ticketId);
        if (ticket is null)
            throw new KeyNotFoundException($"Ticket '{ticketId}' not found.");

        return ticket;
    }

    private async Task<Sla> GetSlaOrThrowAsync(Ticket ticket)
    {
        var sla = await _slaRepository.GetByIdAsync(ticket.SlaId);
        if (sla is null)
            throw new KeyNotFoundException($"SLA '{ticket.SlaId}' referenced by ticket '{ticket.TicketId}' was not found.");

        return sla;
    }

        var existing = await _slaRepository.GetByTenantAndPriorityAsync(tenantId, dto.Priority);
        if (existing is not null)

        return _mapper.Map<SlaResponseDto>(sla);
    }

    {
        var tenantId = _currentTenantService.GetTenantId();
       
        var sla = await _slaRepository.GetByTenantAndPriorityAsync(tenantId, priority);

        if (sla == null){
            throw new KeyNotFoundException($"No SLA configured for priority '{priority}'.");
        }

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task<IEnumerable<SlaResponseDto>> GetAllAsync()
    {
        var tenantId = _currentTenantService.GetTenantId();
        var slas = await _slaRepository.GetAllByTenantAsync(tenantId);

        return _mapper.Map<IEnumerable<SlaResponseDto>>(slas);
    }

    public async Task<SlaResponseDto> UpdateAsync(Guid id, UpdateSlaTimesDto dto)
    {
        if (dto.ResponseTime > dto.ResolutionTime)
            throw new ArgumentException("Response time cannot be greater than resolution time.");

        var sla = await _slaRepository.GetByIdAsync(id);
        EnsureExistsAndOwned(sla);

        sla.UpdateTimes(dto.ResponseTime, dto.ResolutionTime);
        await _slaRepository.UpdateAsync(sla);

        return _mapper.Map<SlaResponseDto>(sla);
    }

    public async Task DeleteAsync(Guid id)
    {
        var sla = await _slaRepository.GetByIdAsync(id);
        EnsureExistsAndOwned(sla);

        await _slaRepository.DeleteAsync(id);
    }

    private void EnsureExistsAndOwned([NotNull] Sla? sla)

        if (sla is null)
            throw new KeyNotFoundException("SLA not found.");

        if (sla.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot access resources belonging to another tenant.");

            ?? throw new KeyNotFoundException($"User with ID '{id}' not found.");

        if (user.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot modify users from another tenant.");

        var normalizedEmail = dto.Email.Trim().ToLower();

        if (!string.Equals(user.Email, normalizedEmail, StringComparison.OrdinalIgnoreCase))

        user.ChangeRole(dto.Role);

using AutoMapper;
using TenantDesk.Application.Dto.Tenant;
using TenantDesk.Application.Interfaces;
using TenantDesk.Domain.Entities;

namespace TenantDesk.Application.Services;

public class TenantService(ITenantRepository tenantRepository,IPlanRepository planRepository,IMapper mapper)
{
    private readonly ITenantRepository _tenantRepository = tenantRepository;
    private readonly IPlanRepository _planRepository = planRepository;
    private readonly IMapper _mapper = mapper;

    public async Task<TenantResponseDto> CreateAsync(CreateTenantDto dto)
    {
        var plan = await _planRepository.GetByIdAsync(dto.PlanId);

        if (plan is null)
            throw new KeyNotFoundException("Plan not found.");

        var subdomainExists = await _tenantRepository.ExistsBySubdomainAsync(dto.Subdomain.Trim().ToLowerInvariant());

        if (subdomainExists)
            throw new InvalidOperationException("Subdomain is already in use.");

        var tenant = new Tenant(dto.PlanId,dto.TenantName,dto.Subdomain);

        await _tenantRepository.AddAsync(tenant);

        return _mapper.Map<TenantResponseDto>(tenant);
    }

    public async Task<TenantResponseDto?> GetByIdAsync(Guid id)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException($"Tenant with ID '{id}' not found.");

        return _mapper.Map<TenantResponseDto>(tenant);
    }

    public async Task<IEnumerable<TenantListDto>> GetAllAsync()
    {
        var tenants = await _tenantRepository.GetAllAsync();

        return _mapper.Map<IEnumerable<TenantListDto>>(tenants);
    }

    public async Task<TenantResponseDto> UpdateAsync(Guid id,UpdateTenantDto dto)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException("Tenant not found.");

        var normalizedSubdomain =
            dto.Subdomain.Trim().ToLowerInvariant();

        if (!string.Equals(tenant.Subdomain,normalizedSubdomain,StringComparison.OrdinalIgnoreCase))
        {
            var subdomainExists = await _tenantRepository.ExistsBySubdomainAsync(normalizedSubdomain);

            if (subdomainExists)
                throw new InvalidOperationException("Subdomain is already in use.");
        }

        tenant.Rename(dto.TenantName);
        tenant.ChangeSubdomain(normalizedSubdomain);

        await _tenantRepository.UpdateAsync(tenant);

using Microsoft.Extensions.Caching.Distributed;
using Microsoft.Extensions.Logging;
using System.Text.Json;
using TenantDesk.Application.Dto.Dashboard;
using TenantDesk.Application.Interfaces.Identity;
using TenantDesk.Domain.Enums;

namespace TenantDesk.Application.Services;

public class DashboardService(
    ITicketRepository _ticketRepository,
    ICurrentTenantService _currentTenantService,
    IDistributedCache _cache,
    ILogger<DashboardService> _logger)
{
    public async Task<DashboardResponseDto> GetAsync()
    {
        var tenantId = _currentTenantService.GetTenantId();

        var cacheKey = $"dashboard:{tenantId}";

        var cachedDashboard = await _cache.GetStringAsync(cacheKey);

        if (cachedDashboard is not null)
        {
            _logger.LogInformation(
                "Dashboard cache HIT for tenant {TenantId}",
                tenantId);

            return JsonSerializer.Deserialize<DashboardResponseDto>(
                cachedDashboard)!;
        }

        _logger.LogInformation(
            "Dashboard cache MISS for tenant {TenantId}",
            tenantId);

        var totalTickets =
            await _ticketRepository.CountByTenantAsync(tenantId);

        var openTickets =
            await _ticketRepository.CountByStatusAsync(
                tenantId,
                TicketStatus.Open);

        var assignedTickets =
            await _ticketRepository.CountByStatusAsync(
                tenantId,
                TicketStatus.Assigned);

        var inProgressTickets =
            await _ticketRepository.CountByStatusAsync(
                tenantId,
                TicketStatus.InProgress);

        var closedTickets =
            await _ticketRepository.CountByStatusAsync(
                tenantId,
                TicketStatus.Closed);

        var dashboard = new DashboardResponseDto(
            totalTickets,
            openTickets,
            assignedTickets,
            inProgressTickets,
            closedTickets
        );

        var serializedDashboard =
            JsonSerializer.Serialize(dashboard);

        await _cache.SetStringAsync(
            cacheKey,
            serializedDashboard,
            new DistributedCacheEntryOptions
            {
                SlidingExpiration =
                    TimeSpan.FromMinutes(5)
            });

        return dashboard;

using AutoMapper;

namespace TenantDesk.Application.Services;

public class CategoryService(ICategoryRepository _categoryRepository, ICurrentTenantService _currentTenantService, IMapper _mapper)
{
    public async Task<CategoryResponseDto> GetByIdAsync(Guid id)
    {
        var category = await _categoryRepository.GetByIdAsync(id)
            ?? throw new KeyNotFoundException($"Category with ID '{id}' not found.");

        if (category.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot access categories from another tenant.");

        return _mapper.Map<CategoryResponseDto>(category);
    }

    public async Task<PagedResult<CategoryResponseDto>> GetFilteredAsync(CategoryFilter filter)
    {
        filter.TenantId = _currentTenantService.GetTenantId();

        var result = await _categoryRepository.GetFilteredAsync(filter);

        return new PagedResult<CategoryResponseDto>
        {
            Items = result.Items.Select(c => _mapper.Map<CategoryResponseDto>(c)).ToList(),
            TotalCount = result.TotalCount,
            Page = result.Page,
            PageSize = result.PageSize
        };
    }

    public async Task<CategoryResponseDto> CreateAsync(CreateCategoryDto dto)
    {
        var tenantId = _currentTenantService.GetTenantId();

        var nameTaken = await _categoryRepository.ExistsByNameAsync(tenantId, dto.CategoryName);
        if (nameTaken)
            throw new InvalidOperationException("A category with this name already exists in this tenant.");

        var category = new Category(tenantId, dto.CategoryName, dto.Description);

        await _categoryRepository.AddAsync(category);

        return _mapper.Map<CategoryResponseDto>(category);
    }

    public async Task<CategoryResponseDto> UpdateAsync(Guid id, UpdateCategoryDto dto)
    {
        var category = await _categoryRepository.GetByIdAsync(id)
            ?? throw new KeyNotFoundException($"Category with ID '{id}' not found.");

        if (category.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot modify categories from another tenant.");

        if (!string.Equals(category.CategoryName, dto.CategoryName, StringComparison.OrdinalIgnoreCase))
        {
            var nameTaken = await _categoryRepository.ExistsByNameAsync(category.TenantId, dto.CategoryName);
            if (nameTaken)
                throw new InvalidOperationException("A category with this name already exists in this tenant.");
        }

        category.Rename(dto.CategoryName);
        category.UpdateDescription(dto.Description);

        if (dto.IsActive)
            category.Activate();
        else
            category.Deactivate();

        await _categoryRepository.UpdateAsync(category);

        return _mapper.Map<CategoryResponseDto>(category);
    }

    public async Task DeleteAsync(Guid id)
    {
        var category = await _categoryRepository.GetByIdAsync(id)
            ?? throw new KeyNotFoundException($"Category with ID '{id}' not found.");

        if (category.TenantId != _currentTenantService.GetTenantId())
            throw new UnauthorizedAccessException("You cannot delete categories from another tenant.");

        var ticketCount = await _categoryRepository.GetTicketCountByCategoryAsync(id);
        if (ticketCount > 0)
            throw new InvalidOperationException("Cannot delete a category that still has tickets assigned to it.");

        await _categoryRepository.DeleteAsync(id);
    }
