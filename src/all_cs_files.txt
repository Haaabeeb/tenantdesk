using TenantDesk.Application.Dto.Tenant;
using TenantDesk.Application.Interfaces;
using TenantDesk.Domain.Entities;

namespace TenantDesk.Application.Services;

public class TenantService
{
    private readonly ITenantRepository _tenantRepository;
    private readonly IPlanRepository _planRepository;

    public TenantService(ITenantRepository tenantRepository,IPlanRepository planRepository)
    {
        _tenantRepository = tenantRepository;
        _planRepository = planRepository;
    }

    public async Task<TenantResponseDto> CreateAsync(CreateTenantDto dto)
    {
        var plan = await _planRepository.GetByIdAsync(dto.PlanId);

        if (plan is null)
            throw new KeyNotFoundException("Plan not found.");

        var subdomainExists = await _tenantRepository.ExistsBySubdomainAsync(dto.Subdomain.Trim().ToLowerInvariant());

        if (subdomainExists)
            throw new InvalidOperationException("Subdomain is already in use.");

        var tenant = new Tenant(dto.PlanId,dto.TenantName,dto.Subdomain);

        await _tenantRepository.AddAsync(tenant);

        return MapToResponseDto(tenant);
    }

    public async Task<TenantResponseDto?> GetByIdAsync(Guid id)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null){
            throw new KeyNotFoundException($"Tenant with ID '{id}' not found.");
        }

        return MapToResponseDto(tenant);
    }
    public async Task<IEnumerable<TenantListDto>> GetAllAsync()
    {
        var tenants = await _tenantRepository.GetAllAsync();

        return tenants.Select(t => new TenantListDto(t.TenantId,t.TenantName,t.Subdomain,t.IsActive));
    }

    public async Task<TenantResponseDto> UpdateAsync(Guid id,UpdateTenantDto dto)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException("Tenant not found.");

        var normalizedSubdomain = dto.Subdomain.Trim().ToLowerInvariant();

        if (!string.Equals(tenant.Subdomain,normalizedSubdomain,StringComparison.OrdinalIgnoreCase))
        {
            var subdomainExists = await _tenantRepository.ExistsBySubdomainAsync(normalizedSubdomain);

            if (subdomainExists)
                throw new InvalidOperationException("Subdomain is already in use.");
        }

        tenant.Rename(dto.TenantName);
        tenant.ChangeSubdomain(normalizedSubdomain);

        await _tenantRepository.UpdateAsync(tenant);

        return MapToResponseDto(tenant);
    }

    public async Task ChangePlanAsync(Guid id,ChangeTenantPlanDto dto)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException("Tenant not found.");

        var plan = await _planRepository.GetByIdAsync(dto.PlanId);

        if (plan is null)
            throw new KeyNotFoundException("Plan not found.");

        tenant.ChangePlan(dto.PlanId);

        await _tenantRepository.UpdateAsync(tenant);
    }

    public async Task ChangeStatusAsync(Guid id,ChangeTenantStatusDto dto)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException("Tenant not found.");

        if (dto.IsActive)
            tenant.Activate();
        else
            tenant.Deactivate();

        await _tenantRepository.UpdateAsync(tenant);
    }

    public async Task DeleteAsync(Guid id)
    {
        var tenant = await _tenantRepository.GetByIdAsync(id);

        if (tenant is null)
            throw new KeyNotFoundException("Tenant not found.");

        await _tenantRepository.DeleteAsync(id);
    }

    private static TenantResponseDto MapToResponseDto(Tenant tenant)
    {
        return new TenantResponseDto(tenant.TenantId,tenant.PlanId,tenant.TenantName,tenant.Subdomain,
            tenant.IsActive,tenant.CreatedAt,tenant.UpdatedAt);
    }
}

public class PlanService
{
    private readonly IPlanRepository _repository;

    public PlanService(IPlanRepository repository)
    {
        _repository = repository;
    }

    public async Task<PlanResponseDto> CreateAsync(CreatePlanDto dto)
    {
        var existingPlan = await _repository.GetByNameAsync(dto.Name);

        if (existingPlan is not null)
        {
            throw new InvalidOperationException($"A plan with the name '{dto.Name}' already exists.");
        }

        var plan = new Plan(dto.Name, dto.RateLimitPerMinute);
        
        await _repository.AddAsync(plan);

        return new PlanResponseDto(plan.PlanId, plan.Name, plan.RateLimitPerMinute, plan.CreatedAt);

    }

    public async Task<PlanResponseDto> ChangeRateLimitAsync(Guid planId, ChangePlanRateLimitDto dto)
    {
        var plan = await _repository.GetByIdAsync(planId);

        if (plan is null)
        {
            throw new KeyNotFoundException($"Plan with ID '{planId}' not found.");
        }

        plan.ChangeRateLimit(dto.RateLimitPerMinute);
       
        await _repository.UpdateAsync(plan);

        return new PlanResponseDto(plan.PlanId, plan.Name, plan.RateLimitPerMinute, plan.CreatedAt);
    }

    {
        var plan = await _repository.GetByIdAsync(planId);

        if (plan is null)

namespace TenantDesk.Infrastructure.Identity;

public class AuthService : IAuthService
{
    private readonly IUserRepository _userRepository;
    private readonly ITenantRepository _tenantRepository;
    private readonly ITokenService _tokenService;
    private readonly IPasswordHasher<User> _passwordHasher;
    private readonly IUnitOfWork _unitOfWork;

    public AuthService(
        IUserRepository userRepository,
        ITenantRepository tenantRepository,
        ITokenService tokenService,
        IPasswordHasher<User> passwordHasher,
        IUnitOfWork unitOfWork)
    {
        _userRepository = userRepository;
        _tenantRepository = tenantRepository;
        _tokenService = tokenService;
        _passwordHasher = passwordHasher;
        _unitOfWork = unitOfWork;
    }

    public async Task<AuthResponseDto> RegisterAsync(RegisterDto dto)
    {
        // 1. Resolve the tenant from the subdomain the frontend sent.
        //    If it doesn't exist, we can't know which tenant to attach
        //    this user to, so we stop here.
        var tenant = await _tenantRepository.GetBySubdomainAsync(dto.TenantSubdomain)
            ?? throw new KeyNotFoundException("Tenant not found.");

        // 2. Prevent duplicate accounts WITHIN this tenant. Per-tenant
        //    uniqueness means the same email CAN exist in different tenants,
        //    but not twice in the same one — that's exactly what
        //    ExistsByEmailAsync(tenantId, email) checks.
        var emailTaken = await _userRepository.ExistsByEmailAsync(tenant.TenantId, dto.Email.Trim().ToLower());
        if (emailTaken)
            throw new InvalidOperationException("A user with this email already exists in this tenant.");

        // 3. Create the user WITHOUT a password (see Step 2 change to User.cs),
        //    then hash the real password and attach it explicitly.
        //    Default role here is Customer — change this if your business
        //    rule is different (e.g. invite-only admins vs self-signup customers).
        var user = new User(tenant.TenantId, dto.Email, UserRole.Customer);
        var hash = _passwordHasher.HashPassword(user, dto.Password);
        user.SetPassword(hash);

        // 4. Persist to the database. AddAsync stages the change in the
        //    DbContext; the actual SQL INSERT + commit happens in
        //    _unitOfWork.SaveChangesAsync() — this is the pattern your
        //    UnitOfWork class already uses everywhere else in the project.
        await _userRepository.AddAsync(user);
        await _unitOfWork.SaveChangesAsync();

        // 5. Issue a token immediately so the user is logged in right after
        //    registering (no separate login step required).
        var token = _tokenService.TokenGenerateToken(user);
        var expiresAt = GetExpiryFromToken(token);
        return new AuthResponseDto(token, user.UserId, user.TenantId, user.Email, user.Role, expiresAt);
    }

    public async Task<AuthResponseDto> LoginAsync(LoginDto dto)
    {
        // Resolve tenant the same way as registration.
        var tenant = await _tenantRepository.GetBySubdomainAsync(dto.TenantSubdomain)
            ?? throw new UnauthorizedAccessException("Invalid credentials.");

        // Tenant-scoped lookup: finds the user only if they belong to THIS
        // tenant, so there's no ambiguity if the same email exists under a
        // different tenant.
        var user = await _userRepository.GetByEmailAsync(dto.Email.Trim().ToLower(), tenant.TenantId)
            ?? throw new UnauthorizedAccessException("Invalid credentials.");

        // Note: we deliberately throw the SAME generic "Invalid credentials"
        // message whether the tenant, email, or password is wrong. This
        // stops attackers from figuring out (e.g.) "this email exists but
        // the password is wrong" — a common security leak in login endpoints.

        if (!user.IsActive)
            throw new UnauthorizedAccessException("This account has been deactivated.");

        var result = _passwordHasher.VerifyHashedPassword(user, user.PasswordHash, dto.Password);
        if (result == PasswordVerificationResult.Failed)
            throw new UnauthorizedAccessException("Invalid credentials.");

        var token = _tokenService.TokenGenerateToken(user);
        var expiresAt = GetExpiryFromToken(token);
        return new AuthResponseDto(token, user.UserId, user.TenantId, user.Email, user.Role, expiresAt);
    }

    // ITokenService only returns the raw JWT string, not an expiry value
    // separately. Rather than recompute/duplicate the expiry duration here
    // (which could drift out of sync with whatever TokenService actually
    // set inside the token), we read it straight out of the token's own
    // "exp" claim. The token itself stays the single source of truth.
    private static DateTime GetExpiryFromToken(string token)
    {
        var handler = new JwtSecurityTokenHandler();
        var jwt = handler.ReadJwtToken(token);
        return jwt.ValidTo; 
    }

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtSection["Key"]!));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

            issuer: jwtSection["Issuer"],
            audience: jwtSection["Audience"],
            claims: claims,
            expires: expiresAt,
            signingCredentials: creds
        );
